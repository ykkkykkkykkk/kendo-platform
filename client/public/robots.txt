# robots.txt의 Disallow는 '앞부분이 같으면' 걸린다. 그래서 /player 하나를 막으려고
# 'Disallow: /player'라고 쓰면 /players/hong-seonghun 같은 선수 프로필까지 전부 막힌다
# (실제로 그렇게 막혀서 구글이 선수 페이지를 못 읽었다).
# 한 주소만 막을 때는 반드시 '$'로 끝을 못박는다.

User-agent: *
Allow: /

# 선수 프로필은 이 사이트에서 제일 중요한 공개 페이지다.
# 아래 /player 규칙에 다시 휩쓸리지 않도록 명시해 둔다
# (규칙이 겹치면 더 길게 일치하는 쪽이 이긴다).
Allow: /players
Allow: /players/

# 관리자 화면 (/admin, /admin/… 전부)
Disallow: /admin

# 선수 본인 전용 화면 — '/player' 딱 하나. '$'가 없으면 /players/…까지 막힌다.
Disallow: /player$
Disallow: /player/

# 내 정보 — '/me'와 그 하위만. '$'가 없으면 나중에 /media 같은 주소가 생겼을 때 같이 막힌다.
Disallow: /me$
Disallow: /me/

# 로그인 콜백 — 사람이 볼 페이지가 아니다
Disallow: /oauth/

# 검색 결과는 볼 때마다 내용이 달라 색인할 값이 없다
Disallow: /search

# 개발용 화면
Disallow: /debug

# API 응답은 페이지가 아니다
Disallow: /api/

Sitemap: https://www.minorstar.kr/sitemap.xml
